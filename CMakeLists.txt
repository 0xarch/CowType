    file(GLOB VOICE_CRATES LIST_DIRECTORIES true "${CMAKE_SOURCE_DIR}/build/voice-vendor/*")
    foreach(CRATE IN LISTS VOICE_CRATES)
        get_filename_component(CRATE_NAME "${CRATE}" NAME)
        file(GLOB_RECURSE CRATE_NOTICES "${CRATE}/*LICENSE*" "${CRATE}/*COPYING*" "${CRATE}/*NOTICE*")
        foreach(NOTICE IN LISTS CRATE_NOTICES)
            if(NOT IS_DIRECTORY "${NOTICE}")
                file(RELATIVE_PATH NOTICE_REL "${CRATE}" "${NOTICE}")
                get_filename_component(NOTICE_DIR "${NOTICE_REL}" DIRECTORY)
                install(FILES "${NOTICE}" DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/cowtype/voice-dependencies/${CRATE_NAME}/${NOTICE_DIR}")

        endforeach()
    endforeach()
