cmake_minimum_required(VERSION 3.20)
project(CowType VERSION 0.1.0 LANGUAGES C CXX)

install(FILES src-settings/i18n/catalog.py DESTINATION ${CMAKE_INSTALL_LIBDIR}/cowtype/i18n)

file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/*.rs")
add_custom_command(OUTPUT "${RUST_TARGET}/release/libcowtype.a" "${RUST_TARGET}/release/cowtype"
    COMMAND ${CMAKE_COMMAND} -E env "CARGO_TARGET_DIR=${RUST_TARGET}" ${CARGO_EXECUTABLE} build --release --offline --locked --features native-ffi
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" DEPENDS ${RUST_SOURCES} ${I18N_RESOURCES} Cargo.toml Cargo.lock VERBATIM)
add_custom_target(cowtype-rust ALL DEPENDS "${RUST_TARGET}/release/libcowtype.a" "${RUST_TARGET}/release/cowtype")
add_library(cowtype-core STATIC IMPORTED GLOBAL)
set_target_properties(cowtype-core PROPERTIES IMPORTED_LOCATION "${RUST_TARGET}/release/libcowtype.a")
add_dependencies(cowtype-core cowtype-rust)
add_library(cowtype-fcitx5 MODULE src-fcitx5/engine/cowtype.cpp)
target_include_directories(cowtype-fcitx5 PRIVATE src-fcitx5/include "${CMAKE_BINARY_DIR}")
add_dependencies(cowtype-fcitx5 cowtype-i18n)
target_compile_definitions(cowtype-fcitx5 PRIVATE COWTYPE_SYSTEM_DATA_DIR="${CMAKE_INSTALL_FULL_DATADIR}/cowtype/dictionaries" COWTYPE_VOICE_EXECUTABLE="${CMAKE_INSTALL_FULL_BINDIR}/cowtype-voice")
target_compile_options(cowtype-fcitx5 PRIVATE -Wall -Wextra -Werror)
target_link_libraries(cowtype-fcitx5 PRIVATE cowtype-core LibIME::Pinyin Fcitx5::Core Threads::Threads ${CMAKE_DL_LIBS} m)
configure_file(src-fcitx5/config/cowtype-addon.conf.in cowtype-addon.conf @ONLY)
install(TARGETS cowtype-fcitx5 LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/fcitx5)
install(FILES "${CMAKE_BINARY_DIR}/cowtype-addon.conf" DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/addon RENAME cowtype.conf)
install(FILES src-fcitx5/config/cowtype.conf src-fcitx5/config/cowtype-wubi.conf DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/inputmethod)
install(PROGRAMS "${RUST_TARGET}/release/cowtype" DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY "${COWTYPE_DICTIONARY_DIR}/ice" "${COWTYPE_DICTIONARY_DIR}/wubi" DESTINATION ${CMAKE_INSTALL_DATADIR}/cowtype/dictionaries)
install(DIRECTORY assets/icons/hicolor/ DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/cowtype)
install(DIRECTORY src-data-tools/manifest/notices/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/cowtype/dictionary-notices)
install(FILES src-data-tools/dictionaries/manage.py DESTINATION ${CMAKE_INSTALL_LIBDIR}/cowtype/dictionaries)
install(FILES src-data-tools/manifest/sources.json DESTINATION ${CMAKE_INSTALL_LIBDIR}/cowtype/manifest)
install(FILES src-settings/app/main.py src-settings/app/launch.py DESTINATION ${CMAKE_INSTALL_LIBDIR}/cowtype/settings)
configure_file(src-settings/desktop/cowtype-config.in cowtype-config @ONLY)
configure_file(src-data-tools/cli/cowtype-dictionaries.in cowtype-dictionaries @ONLY)
install(PROGRAMS "${CMAKE_BINARY_DIR}/cowtype-config" "${CMAKE_BINARY_DIR}/cowtype-dictionaries" DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES src-settings/desktop/io.github._0xarch.CowType.Config.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)

# Preserve upstream notices for compiled voice dependencies as well as full source archive.
if(EXISTS "${CMAKE_SOURCE_DIR}/build/voice-vendor")
    file(GLOB VOICE_CRATES LIST_DIRECTORIES true "${CMAKE_SOURCE_DIR}/build/voice-vendor/*")
    foreach(CRATE IN LISTS VOICE_CRATES)
        get_filename_component(CRATE_NAME "${CRATE}" NAME)
        file(GLOB_RECURSE CRATE_NOTICES "${CRATE}/*LICENSE*" "${CRATE}/*COPYING*" "${CRATE}/*NOTICE*")
        foreach(NOTICE IN LISTS CRATE_NOTICES)
            if(NOT IS_DIRECTORY "${NOTICE}")
                file(RELATIVE_PATH NOTICE_REL "${CRATE}" "${NOTICE}")
                get_filename_component(NOTICE_DIR "${NOTICE_REL}" DIRECTORY)
                install(FILES "${NOTICE}" DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/cowtype/voice-dependencies/${CRATE_NAME}/${NOTICE_DIR}")
            endif()
        endforeach()
    endforeach()
endif()
